// Exercise 2

//dp[i] = max value gained w/activities i -> N-1
solve-wa():
	create dp[N+1]
	initialize dp[] to -1

	//base cases
	dp[N] = 0, dp[N-1] = v[N-1]

	solve(0, dp)
	return dp[0]

solve(i, dp):
	if(dp[i] != -1) return dp[i]

	dontTake = solve(i+1, dp)
	take = v[i] + solve(NEXT(i), dp)
	dp[i] = max(dontTake, take)

	return dp[i]
	
NEXT(i): 
	f = i+1, l = N-1
	ans = N
	while(f <= l):
		mid = (f+l) / 2
		if(A[mid].start >= A[i].finish):
			ans = mid
			// search left for an earlier compatible activity
			l = mid - 1
		else:
			// activity starts too early, search right  
      f = mid + 1

	return ans



// Exercise 3
/*
  - optimal substructure:
	optimal(i) = {
			            0						                                    if i < 0
			            MAX(optimal(PREV(i)) + v[i], optimal(i-1))	    if i >= 0
		           }

  - base cases:
	dp[0] = v[0]

  - final soltuion:
	dp[N-1]

  - order of traversal:
	0 .. N-1
*/

// we sort the activities by finish time because dp[i] is the optimal solution considering activities 0 to i and we need dp[PREV(i)]
// PREV(i) searches for the largest j such that A[j].finish <= A[i].start 
// PREV(i) must be an index before i, sorting by finish time ensures that every compatible activity with i appears somewhere in [0, i-1] 

solve-wa():
	sort A[] based on finish time
	create dp[N]
	dp[0] = v[0]

	for(i = 1; i < N; i++):
		dontTake = dp[i-1]
		take = v[i]
		if(PREV(i) != -1) 
      take += dp[PREV(i)]
		dp[i] = max(dontTake, take)

	return dp[N-1]


PREV(i):
	// search for the last compatible previous activity
	f = 0, l = i-1
	ans = -1
	while(f <= l):
		mid = (f+l) / 2
		if(A[mid].finish <= A[i].start):
			ans = mid
			// search right for a later compatible activity 
			f = mid + 1
		else:
			// activity finishes too late, search left 
			l = mid - 1

	return ans
